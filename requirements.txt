MEMBERS = ["DƯƠNG T", "DƯƠNG", "LONG", "XUÂN", "PHƯƠNG", "KHÁNH"]

AUTO_REPORT_HOUR = 23
AUTO_REPORT_MINUTE = 59

def get_vn_now():
    return datetime.now(VN_TZ)

def load_data():
    default_data = {
        "members": MEMBERS,
        "rates": {"dnt": 110, "tha": 130},
        "months": {},
        "group_chat_ids": [],
        "last_auto_report_date": ""
    }
    if not os.path.exists(DATA_FILE):
        return default_data
    try:
        with open(DATA_FILE, "r", encoding="utf-8") as f:
            data = json.load(f)
            for key, val in default_data.items():
                if key not in data:
                    data[key] = val
            return data
    except Exception:
        return default_data

def save_data(data):
    with open(DATA_FILE, "w", encoding="utf-8") as f:
        json.dump(data, f, ensure_ascii=False, indent=2)

def register_chat_id(chat_id):
    data = load_data()
    if chat_id not in data["group_chat_ids"]:
        data["group_chat_ids"].append(chat_id)
        save_data(data)

def add_coin_record(member_name, coin_type, amount, day=None, month=None):
    now = get_vn_now()
    if not day: day = str(now.day)
    else: day = str(day)
    if not month: month = str(now.month)
    else: month = str(month)

    data = load_data()
    
    matched_name = None
    for m in data["members"]:
        if m.lower() == member_name.lower():
            matched_name = m
            break
    if not matched_name:
        matched_name = member_name.upper()
        data["members"].append(matched_name)

    if month not in data["months"]:
        data["months"][month] = {}
    if day not in data["months"][month]:
        data["months"][month][day] = {}
    if matched_name not in data["months"][month][day]:
        data["months"][month][day][matched_name] = {"dnt": 0, "tha": 0}

    data["months"][month][day][matched_name][coin_type] += amount
    save_data(data)
    
    current_val = data["months"][month][day][matched_name][coin_type]
    return matched_name, day, month, current_val

def build_daily_report(day, month):
    data = load_data()
    day_data = data.get("months", {}).get(str(month), {}).get(str(day), {})

    if not day_data:
        return None

    msg = f"📊 <b>BÁO CÁO TỔNG KẾT NGÀY {day}/{month}</b>\n"
    msg += "───────────────────\n"
    total_dnt = total_tha = 0

    for name, coins in day_data.items():
        cdnt = coins.get("dnt", 0)
        ctha = coins.get("tha", 0)
        if cdnt > 0 or ctha > 0:
            total_dnt += cdnt
            total_tha += ctha
            msg += f"• <b>{name}</b>: DNT {cdnt:,} | Thả {ctha:,}\n"

    rate_dnt = data["rates"]["dnt"]
    rate_tha = data["rates"]["tha"]
    total_money = (total_dnt * rate_dnt) + (total_tha * rate_tha)

    msg += "───────────────────\n"
    msg += f"🎁 <b>Tổng Xu DNT:</b> {total_dnt:,} xu ({total_dnt * rate_dnt:,.0f} đ)\n"
    msg += f"🪂 <b>Tổng Xu Thả:</b> {total_tha:,} xu ({total_tha * rate_tha:,.0f} đ)\n"
    msg += f"💰 <b>TỔNG THỰC NHẬN:</b> <b>{total_money:,.0f} VNĐ</b>\n"
    msg += f"🕒 <i>Tự động chốt lúc {get_vn_now().strftime('%H:%M:%S')}</i>"
    return msg

def manual_total(message):

    now = get_vn_now()

@bot.message_handler(func=lambda msg: True)
def handle_message(message):
    register_chat_id(message.chat.id)
    text = message.text.strip().lower()
    parts = text.split()

    if len(parts) < 3:
        return

    name = parts[0]
    try:
        coins = int(parts[1].replace(',', '').replace('.', ''))
    except ValueError:
        return

    type_str = parts[2]
    if "dnt" in type_str or "don" in type_str:
        c_type = "dnt"
        type_name = "Xu Donate"
    elif "tha" in type_str:
        c_type = "tha"
        type_name = "Xu Thả"
    else:
        return

    custom_day = None
    if len(parts) >= 4 and parts[3].isdigit():
        custom_day = int(parts[3])

    try:
        real_name, day, month, current_val = add_coin_record(name, c_type, coins, day=custom_day)
        bot.reply_to(
            message,
            f"✅ <b>Đã nhận:</b> {real_name} (+{coins:,} {type_name})\n"
            f"📅 Ngày {day}/{month} | Tổng hiện tại: <b>{current_val:,} xu</b>",
            parse_mode="HTML"
        )
    except Exception as e:
        bot.reply_to(message, f"❌ Lỗi: {str(e)}")

def auto_scheduler_loop():
    while True:
        try:
            now = get_vn_now()
            today_str = f"{now.day:02d}_{now.month:02d}_{now.year}"

            if now.hour == AUTO_REPORT_HOUR and now.minute == AUTO_REPORT_MINUTE:
                data = load_data()
                if data.get("last_auto_report_date") != today_str:
                    report = build_daily_report(now.day, now.month)
                    if report and data.get("group_chat_ids"):
                        for cid in data["group_chat_ids"]:
                            try:
                                bot.send_message(cid, f"🔔 <b>[TỰ ĐỘNG CHỐT SỔ NGÀY]</b>\n\n{report}", parse_mode="HTML")
                            except Exception as err:
                                print(f"Không gửi được đến chat_id {cid}: {err}")

                    data["last_auto_report_date"] = today_str
                    save_data(data)

            time.sleep(30)
        except Exception as e:
            print(f"Lỗi scheduler: {e}")
            time.sleep(30)

@app.route("/")
def index():
    now = get_vn_now()
    m = str(now.month)
    data = load_data()
    members = data["members"]
    month_data = data.get("months", {}).get(m, {})

    day_totals = {}
    member_totals = {mem: {"dnt": 0, "tha": 0} for mem in members}
    total_m_dnt = total_m_tha = 0

    for d in range(1, 32):
        d_str = str(d)
        d_dnt = d_tha = 0
        for mem in members:
            rec = month_data.get(d_str, {}).get(mem, {"dnt": 0, "tha": 0})
            d_dnt += rec.get("dnt", 0)
            d_tha += rec.get("tha", 0)
            member_totals[mem]["dnt"] += rec.get("dnt", 0)
            member_totals[mem]["tha"] += rec.get("tha", 0)
        day_totals[d] = {"dnt": d_dnt, "tha": d_tha}
        total_m_dnt += d_dnt
        total_m_tha += d_tha

    rates = data.get("rates", {"dnt": 110, "tha": 130})
    total_all_money = (total_m_dnt * rates["dnt"]) + (total_m_tha * rates["tha"])
    total_all_coins = total_m_dnt + total_m_tha

    return render_template_string(
        WEB_HTML,
        current_month=m,
        members=members,
        month_data=month_data,
        day_totals=day_totals,
        member_totals=member_totals,
        total_m_dnt=total_m_dnt,
        total_m_tha=total_m_tha,
        rates=rates,
        total_all_money=total_all_money,
        total_all_coins=total_all_coins
    )
